ถ้าเขียนด้วย JSX (ใน Vite)

function Counter() {
  const [count, setCount] = useState(0);
  return (
    <div>
      <button onClick={() => setCount(count - 1)}>−</button>
      <strong>{count}</strong>
      <button onClick={() => setCount(count + 1)}>+</button>
      {count >= 5 && <p>🎉 ถึง 5 แล้ว!</p>}
    </div>
  );
}
